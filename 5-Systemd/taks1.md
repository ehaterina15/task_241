#!/bin/bash

# =========================================================
# Systemd Units (Юниты)
# =========================================================

# ---------------------------------------------------------
# Что такое systemd-юнит
# ---------------------------------------------------------
# Systemd-юнит — это конфигурационный объект системы инициализации systemd,
# который описывает, как управлять сервисами, устройствами, точками
# монтирования, таймерами и другими ресурсами системы.
#
# Юниты представляют собой файлы с расширениями:
# .service — сервис
# .timer   — таймер
# .mount   — точка монтирования
# .socket  — сокет
# .target  — группа юнитов

# ---------------------------------------------------------
# Проверка статуса systemd-юнита
# ---------------------------------------------------------
# Команда systemctl status выводит:
# - состояние загрузки юнита
# - активен ли он в данный момент
# - включён ли в автозагрузку
# - PID основного процесса
# - последние сообщения журнала

systemctl status sshd || true

# ---------------------------------------------------------
# Остановка сервиса
# ---------------------------------------------------------
sudo systemctl stop sshd || true

# ---------------------------------------------------------
# Перезапуск сервиса
# ---------------------------------------------------------
sudo systemctl restart sshd || true

# ---------------------------------------------------------
# Управление автозагрузкой
# ---------------------------------------------------------
# Удаление сервиса из автозагрузки
sudo systemctl disable sshd || true

# Возврат сервиса в автозагрузку
sudo systemctl enable sshd || true

# ---------------------------------------------------------
# Что такое systemd-таймеры
# ---------------------------------------------------------
# Systemd-таймеры — это механизм планирования задач,
# аналог cron, встроенный в systemd.
#
# Таймер (.timer) определяет, КОГДА выполнять задачу,
# а сервис (.service) определяет, ЧТО выполнять.
#
# Преимущества таймеров:
# - интеграция с systemd
# - централизованные логи
# - обработка пропущенных запусков
# - гибкое управление зависимостями

# =========================================================
# Конец файла
# =========================================================
