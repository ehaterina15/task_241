#!/bin/bash

# =========================================================
# Журналы systemd и SSH
# =========================================================

# ---------------------------------------------------------
# 1. Просмотр журналов службы SSH
# ---------------------------------------------------------
# Команда journalctl с ключом -u позволяет вывести логи
# конкретного systemd-юнита, в данном случае sshd.

sudo journalctl -u sshd

# ---------------------------------------------------------
# 2. Просмотр журналов systemd в реальном времени
# ---------------------------------------------------------
# Ключ -f (follow) включает режим реального времени,
# новые записи появляются сразу после их создания.

sudo journalctl -f

# ---------------------------------------------------------
# 3. Просмотр логов sshd в реальном времени
# ---------------------------------------------------------
# Комбинация -u и -f позволяет следить за событиями
# только одной службы в реальном времени.

sudo journalctl -u sshd -f

# ---------------------------------------------------------
# 4. Можно ли читать логи systemd без journalctl
# ---------------------------------------------------------
# Да, частично можно, если система настроена
# на дублирование логов в текстовые файлы.

# Примеры файлов с логами:
# /var/log/syslog
# /var/log/auth.log

sudo cat /var/log/auth.log || true

# Однако systemd хранит основные журналы в бинарном формате,
# поэтому journalctl является основным и рекомендуемым
# инструментом для работы с логами.

# ---------------------------------------------------------
# 5. Контрольный вопрос
# ---------------------------------------------------------
# Сколько будет 2 - 2?

echo "2 - 2 = 0"

# =========================================================
# Конец файла
# =========================================================
