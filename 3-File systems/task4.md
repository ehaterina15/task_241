#!/bin/bash

# =========================================
# RAID (Redundant Array of Independent Disks)
# =========================================

# RAID — технология объединения нескольких физических дисков
# в один логический массив для повышения скорости и/или надёжности.
# RAID НЕ является резервным копированием.

# -----------------------------------------
# Подготовка виртуальной машины
# -----------------------------------------
# В виртуальную машину добавлены два дополнительных диска.
# Проверяем, как система их видит:
lsblk

# Обычно новые диски отображаются как:
# /dev/sdb
# /dev/sdc

# -----------------------------------------
# Подготовка дисков (разметка)
# -----------------------------------------
# На каждом диске создаётся один раздел под RAID

sudo fdisk /dev/sdb <<EOF
n
p
1


t
fd
w
EOF

sudo fdisk /dev/sdc <<EOF
n
p
1


t
fd
w
EOF

# Проверяем, что разделы созданы
lsblk

# Ожидаемый результат:
# /dev/sdb1
# /dev/sdc1

# -----------------------------------------
# Создание RAID 0
# -----------------------------------------
# RAID 0 — высокая скорость, нет отказоустойчивости

sudo mdadm --create /dev/md0 \
--level=0 \
--raid-devices=2 \
/dev/sdb1 /dev/sdc1

# Проверка состояния RAID
cat /proc/mdstat

# -----------------------------------------
# Форматирование RAID 0 в ext4
# -----------------------------------------
sudo mkfs.ext4 /dev/md0

# Монтирование RAID 0
sudo mkdir -p /mnt/raid0
sudo mount /dev/md0 /mnt/raid0

# Проверка работы RAID 0
cd /mnt/raid0 || exit
touch file1.txt file2.txt
echo "RAID 0 test" > file1.txt
ls
cat file1.txt

# -----------------------------------------
# Удаление RAID 0
# -----------------------------------------
cd /
sudo umount /mnt/raid0
sudo mdadm --stop /dev/md0
sudo mdadm --zero-superblock /dev/sdb1
sudo mdadm --zero-superblock /dev/sdc1

# -----------------------------------------
# Создание RAID 1
# -----------------------------------------
# RAID 1 — полное зеркалирование данных, высокая надёжность

sudo mdadm --create /dev/md1 \
--level=1 \
--raid-devices=2 \
/dev/sdb1 /dev/sdc1

# Проверка состояния RAID
cat /proc/mdstat

# -----------------------------------------
# Форматирование RAID 1 в ext4
# -----------------------------------------
sudo mkfs.ext4 /dev/md1

# Монтирование RAID 1
sudo mkdir -p /mnt/raid1
sudo mount /dev/md1 /mnt/raid1

# Проверка работы RAID 1
cd /mnt/raid1 || exit
touch mirror.txt
echo "RAID 1 test" > mirror.txt
cat mirror.txt

# -----------------------------------------
# Информация
# -----------------------------------------
# RAID 0:
# + высокая скорость
# - отсутствие отказоустойчивости
#
# RAID 1:
# + высокая надёжность
# - полезный объём = объёму одного диска
#
# Альтернативы mdadm:
# - Btrfs (RAID 0/1/10, контроль целостности)
# - ZFS (RAID-Z, серверное применение)
#
# RAID можно создавать во время установки Linux,
# включая установку системы на RAID 1.
