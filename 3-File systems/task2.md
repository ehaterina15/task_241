#!/bin/bash

# =========================
# Структура каталогов Linux
# =========================

# 1. Структура каталогов в Linux
# Linux использует иерархическую структуру, начинающуюся с корня /

# 2. Вывести список файлов и каталогов в корне системы
ls /

# 3. Где хранятся папки пользователей
# Домашние каталоги обычных пользователей находятся в /home
ls /home

# 4. Домашняя папка суперпользователя
ls /root

# 5. Где хранятся основные конфигурационные файлы
ls /etc

# 6. Назначение системных каталогов
# /bin      — базовые пользовательские команды
# /sbin     — системные утилиты администратора
# /usr/bin  — пользовательские программы
# /usr/sbin — административные программы

# Просмотр содержимого каталогов
ls /bin
ls /sbin
ls /usr/bin
ls /usr/sbin
