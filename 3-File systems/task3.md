#!/bin/bash

# =========================================
# Работа с fstab и дисками (Linux)
# =========================================

# 1. Вывести содержимое fstab
# fstab хранит информацию о файловых системах,
# которые должны автоматически монтироваться при загрузке
cat /etc/fstab

# 2. Добавление нового диска
# (выполняется в настройках виртуальной машины, не в консоли)

# 3. Посмотреть, как система видит диски (блочные устройства)
lsblk
# альтернативно
# sudo fdisk -l

# Предположим, новый диск — /dev/sdb

# 4. Создать таблицу разделов и файловую систему ext4
sudo fdisk /dev/sdb <<EOF
n
p
1


w
EOF

# Создание файловой системы ext4
sudo mkfs.ext4 /dev/sdb1

# 5. Примонтировать диск в каталог /mnt
sudo mount /dev/sdb1 /mnt

# 6. Зайти в каталог и создать файлы
cd /mnt || exit
touch file1.txt file2.txt
echo "Test file" > file1.txt
ls

# 7. Отмонтировать диск и проверить файлы
cd /
sudo umount /mnt
ls /mnt

# 8. Настроить автоматическое монтирование через fstab
# Получить UUID диска
sudo blkid /dev/sdb1

# Добавить строку в /etc/fstab (пример):
# UUID=XXXX-XXXX  /mnt  ext4  defaults  0  2

# 9. Проверить корректность fstab без перезагрузки
sudo mount -a

# 10. Перезагрузить систему
sudo reboot

# После загрузки проверить, примонтирован ли диск
# lsblk
# mount | grep /mnt
