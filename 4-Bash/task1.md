#!/bin/bash

# =========================================================
# Скриптование в Linux (Bash)
# =========================================================

# ---------------------------------------------------------
# Что такое шебанг
# ---------------------------------------------------------
# Шебанг — это первая строка скрипта.
# Она указывает, каким интерпретатором должен выполняться файл.
# В данном случае используется bash:
#
#   #!/bin/bash
#
# Благодаря шебангу скрипт можно запускать напрямую: ./script

# ---------------------------------------------------------
# Обязательно ли исполняемому файлу иметь расширение
# ---------------------------------------------------------
# Нет, расширение не обязательно.
# В Linux файл считается исполняемым, если:
# 1) у него есть право на выполнение (chmod +x)
# 2) указан корректный шебанг
#
# Расширение .sh — это лишь соглашение для удобства.

# ---------------------------------------------------------
# Режимы отладки скрипта
# ---------------------------------------------------------
# set -euo pipefail включает:
#
# -e  — немедленно завершает скрипт при любой ошибке
# -u  — ошибка при использовании необъявленных переменных
# -o pipefail — ошибка в любой части пайпа считается ошибкой всей команды
#
# Эти флаги делают скрипт более надёжным и удобным для отладки

set -euo pipefail

# =========================================================
# Работа с файлами (автоматически)
# =========================================================

# Создание папки с подпапками
mkdir -p work_dir/sub_dir

# Переход в директорию
cd work_dir

# Создание файла и запись содержимого
echo "Первая строка" > file.txt
echo "Вторая строка" >> file.txt

# Вывод содержимого файла
cat file.txt

# Копирование файла
cp file.txt copy.txt

# Перемещение файла в подпапку
mv copy.txt sub_dir/

# Переименование файла
mv file.txt renamed.txt

# Сравнение файлов
# diff вернёт ненулевой код, поэтому добавлен || true
diff renamed.txt sub_dir/copy.txt || true

# Сортировка содержимого файла
# По возрастанию
sort renamed.txt

# По убыванию
sort -r renamed.txt

# Удаление всех созданных файлов и папок
cd ..
rm -r work_dir

# =========================================================
# Конец скрипта
# =========================================================
